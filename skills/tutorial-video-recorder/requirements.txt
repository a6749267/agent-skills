# 安装:
#   pip install -r requirements.txt
#   playwright install chromium

#   Pillow                     黑场像素占比检测 / 边缘密度离群检测
#   numpy                      数值运算
#   psycopg2-binary            灌数据库业务数据(可选)
#
# 外部服务:
#   TTS: MiniMax api.minimaxi.com/v1/t2a_v2
#        env: MINIMAX_CN_API_KEY
